1. ログイン認証の要点 (Login.tsx)

SupabaseのAuth機能を活用し、セキュアなログインフローを実現しました。

// ログイン処理：Supabase Auth連携
const handleLogin = async (e: React.FormEvent) => {
  e.preventDefault();
  const { error } = await supabase.auth.signInWithPassword({ email, password });
  if (error) {
    alert("ログイン失敗: " + error.message); // ユーザーへのフィードバック
  } else {
    navigate('/'); // 認証成功時の遷移
  }
};
